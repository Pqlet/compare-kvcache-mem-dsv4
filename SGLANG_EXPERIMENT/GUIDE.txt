

python3 -m venv .venv-sglang

# --- git clone ---
env \
  -u GIT_ASKPASS \
  -u GIT_ASKPASS_DOMAIN \
  -u SSH_ASKPASS \
  -u VSCODE_GIT_ASKPASS_NODE \
  -u VSCODE_GIT_ASKPASS_MAIN \
  -u VSCODE_GIT_ASKPASS_EXTRA_ARGS \
  GIT_TERMINAL_PROMPT=0 \
  git -c credential.helper= clone https://github.com/sgl-project/sglang.git
cd sglang
git checkout a1fe4e30

# --- install packages into env ---
SGLANG_BUILD_RUST_EXTS=none pip install -e python

# --- check versions ---
python -c "import sglang; print(sglang.__version__)"
python -m sglang.check_env

# --- Old cubins from base image stayed installed. New FlashInfer rejects them. ---
python -m pip uninstall -y flashinfer-cubin flashinfer-jit-cache
flashinfer install-cubin-wheel
flashinfer install-jit-cache-wheel   # optional, cuts JIT startup time
python -m sglang.check_env | grep flashinfer
